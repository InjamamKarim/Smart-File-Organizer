def fibonacci(n):
    """Return the nth Fibonacci number."""
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

# Example usage
if __name__ == "__main__":
    for i in range(10):
        print(f"fib({i}) = {fibonacci(i)}")

# Time complexity: O(n)
# Space complexity: O(1)
